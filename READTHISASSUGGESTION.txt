from huggingface_hub import snapshot_download, hf_hub_download
from transformers import AutoTokenizer, AutoModelForCausalLM, pipeline
import os

def download_model(repo_id: str, local_dir: str = None) -> str:
    """
    Download the full model repo (or necessary files) from Hugging Face,
    return the local path where it's stored.
    """
    # If user gave a local path instead of repo_id
    if os.path.isdir(repo_id):
        return repo_id

    # local_dir = folder where we want to store the model
    if local_dir is None:
        local_dir = os.path.join("models", repo_id.replace("/", "_"))

    # If already downloaded
    if os.path.exists(local_dir):
        return local_dir

    # download snapshot (all files) to local_dir
    print(f"Downloading model {repo_id} from Hugging Face...")
    snapshot_download(repo_id=repo_id, local_dir=local_dir, repo_type="model")
    return local_dir

def load_model_and_tokenizer(model_source: str):
    """
    Given model_source (either local folder or hub repo_id),
    returns (tokenizer, model, text_generation_pipeline)
    """
    # Step 1: download / get local path
    local_model_path = download_model(model_source)
    
    # Step 2: load tokenizer & model
    tokenizer = AutoTokenizer.from_pretrained(local_model_path, local_files_only=True)
    model = AutoModelForCausalLM.from_pretrained(local_model_path, local_files_only=True)
    
    # Step 3: build pipeline
    text_gen = pipeline("text-generation", model=model, tokenizer=tokenizer)
    
    return tokenizer, model, text_gen

def ask(model_id: str, prompt: str, **kwargs):
    tokenizer, model, text_gen = load_model_and_tokenizer(model_id)
    results = text_gen(prompt, **kwargs)
    return results

if __name__ == "__main__":
    # Example usage:
    model_name = "gpt2"  # user-chosen or default
    prompt = "Hello, how are you?"
    response = ask(model_name, prompt, max_new_tokens=50)
    print(response)

def list_cached_models(models_folder="models"):
    if not os.path.exists(models_folder):
        return []
    return [d for d in os.listdir(models_folder) if os.path.isdir(os.path.join(models_folder, d))]

def switch_model():
    cached = list_cached_models()
    if not cached:
        print("⚠️ No cached models found. Run first-time setup first.")
        return None

    print("\n📂 Cached Models:")
    for i, m in enumerate(cached, 1):
        print(f"{i}. {m}")

    while True:
        try:
            choice = int(input("Enter number to switch model: "))
            if 1 <= choice <= len(cached):
                selected = cached[choice - 1]
                print(f"✅ Switched to model: {selected}")
                return selected
            else:
                print("Invalid choice. Try again.")
        except ValueError:
            print("Enter a valid number.")
